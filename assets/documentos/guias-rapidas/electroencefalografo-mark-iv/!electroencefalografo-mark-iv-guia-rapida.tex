% ---------- MONTAJE ----------
\seccionbanda{MONTAJE}
\noindent\colorbox{lightblue}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
\begin{itemize}[label=$\square$, itemsep=4pt]
\item Verificar estado físico del equipo.
\item Conectar batería.
\item Conectar dongle a la computadora y verificar comunicación.
\item Encender equipo.
\item Colocar el casco alineando el nodo central posterior con el inion y el nodo frontal con el puente nasal.
\item Para ajustar el casco, desconecte uno a uno cada jumper de conexión y luego vuelva a conectarlo; esto evita dañar la soldadura por torsión y reduce errores de reconexión.
\item Ajustar las unidades de confort para distribuir el peso del casco.
\item Verificar que los electrodos hagan contacto con el cuero cabelludo.
\item Conectar los clips de oreja.
\item Confirmar en el software las señales correspondientes a cada canal.
\end{itemize}
}}\par

% ---------- DESMONTAJE ----------
\seccionbanda{DESMONTAJE}
\noindent\colorbox{lightblue}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
\begin{itemize}[label=$\square$, itemsep=4pt]
\item Apagar el equipo.
\item Para retirar el casco, desconecte uno a uno cada jumper de conexión y luego vuelva a conectarlo después de aflojarlo.
\item Limpiar electrodos e inspeccionar desgaste, puntas dobladas o partes rotas.
\item Desconectar el dongle de la computadora.
\item Ordenar los cables siguiendo la forma de la fibra para evitar quiebres.
\item Retirar las baterías de los bloques para batería.
\item Dejar todo en su lugar.
\end{itemize}
}}\par
