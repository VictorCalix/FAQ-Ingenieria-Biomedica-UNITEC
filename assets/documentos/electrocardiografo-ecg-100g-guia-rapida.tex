% ---------- VERIFICACIÓN DE ELEMENTOS ----------
\seccionbanda{VERIFICACIÓN DE ELEMENTOS}
\noindent\colorbox{lightblue}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
\begin{itemize}[label=$\square$, itemsep=4pt]
\item Equipo en buen estado físico
\item Cable de paciente en buen estado
\item Electrodos funcionales y limpios
\item Gel conductor disponible
\item Papel térmico instalado
\item Alimentación eléctrica segura o batería cargada
\item Derivaciones correctamente identificadas y conectadas
\end{itemize}
}}\par

% ---------- MONTAJE ----------
\seccionbanda{MONTAJE}
\noindent\colorbox{lightblue}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
\begin{itemize}[label=$\square$, itemsep=4pt]
\item Verificar estado físico del equipo.
\item Conectar a red eléctrica.
\item Encender equipo y confirmar autochequeo correcto.
\item Confirmar en pantalla estado de derivaciones ``NORM''.
\item Colocar electrodos de extremidades (RA, LA, RL, LL).
\item Verificar que los electrodos no estén en contacto entre sí.
\item Colocar electrodos precordiales (V1--V6).
\end{itemize}
}}\par

% ---------- DESMONTAJE ----------
\seccionbanda{DESMONTAJE}
\noindent\colorbox{lightblue}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
\begin{itemize}[label=$\square$, itemsep=4pt]
\item Retiro de electrodos.
\item Desconexión de accesorios.
\item Limpieza de material reutilizable.
\item Orden de cables, al enrollarlos, procura seguir la forma de la fibra para evitar quebrarla
\item Apagar el equipo.
\item Dejar todo en su lugar.
\end{itemize}
}}\par
